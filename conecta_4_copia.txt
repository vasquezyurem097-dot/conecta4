#include <stdio.h>

#define FILAS 6
#define COLUMNAS 7

char tablero[FILAS][COLUMNAS];

void inicializarTablero()
{
    int i, j;
    for(i = 0; i < FILAS; i++)
    {
        for(j = 0; j < COLUMNAS; j++)
        {
            tablero[i][j] = '.';
        }
    }
}

void mostrarTablero()
{
    int i, j;
    for(i = 0; i < FILAS; i++)
    {
        for(j = 0; j < COLUMNAS; j++)
        {
            printf("| %c ", tablero[i][j]);
        }
        printf("|\n");
    }
    printf("  1  2  3  4  5  6  7\n");
}

int ponerFicha(int columna, char ficha)
{
    int i;
    if(columna < 0 || columna >= COLUMNAS)
    {
        printf("Columna invalida.\n");
        return 0;
    }

    for(i = FILAS - 1; i >= 0; i--)
    {
        if(tablero[i][columna] == '.')
        {
            tablero[i][columna] = ficha;
            return 1;
        }
    }

    printf("La columna esta llena.\n");
    return 0;
}

int revisarGanador(char ficha)
{
    int i, j;

    // Horizontal
    for(i = 0; i < FILAS; i++)
        for(j = 0; j < COLUMNAS - 3; j++)
            if(tablero[i][j] == ficha &&
               tablero[i][j+1] == ficha &&
               tablero[i][j+2] == ficha &&
               tablero[i][j+3] == ficha)
                return 1;

    // Vertical
    for(i = 0; i < FILAS - 3; i++)
        for(j = 0; j < COLUMNAS; j++)
            if(tablero[i][j] == ficha &&
               tablero[i+1][j] == ficha &&
               tablero[i+2][j] == ficha &&
               tablero[i+3][j] == ficha)
                return 1;

    // Diagonal hacia abajo
    for(i = 0; i < FILAS - 3; i++)
        for(j = 0; j < COLUMNAS - 3; j++)
            if(tablero[i][j] == ficha &&
               tablero[i+1][j+1] == ficha &&
               tablero[i+2][j+2] == ficha &&
               tablero[i+3][j+3] == ficha)
                return 1;

    // Diagonal hacia arriba
    for(i = 3; i < FILAS; i++)
        for(j = 0; j < COLUMNAS - 3; j++)
            if(tablero[i][j] == ficha &&
               tablero[i-1][j+1] == ficha &&
               tablero[i-2][j+2] == ficha &&
               tablero[i-3][j+3] == ficha)
                return 1;

    return 0;
}

int main()
{
    int columna;
    int jugador = 1;
    int terminado = 0;
    char ficha;

    inicializarTablero();

    while(terminado == 0)
    {
        mostrarTablero();

        if(jugador == 1)
        {
            ficha = 'X';
            printf("Turno del jugador 1 (X)\n");
        }
        else
        {
            ficha = 'O';
            printf("Turno del jugador 2 (O)\n");
        }

        printf("Ingresa una columna del 1 al 7: ");

        if(scanf("%d", &columna) != 1)
        {
            printf("Entrada invalida. Fin del juego.\n");
            return 1;
        }

        columna = columna - 1;

        if(ponerFicha(columna, ficha) == 1)
        {
            if(revisarGanador(ficha) == 1)
            {
                mostrarTablero();
                printf("Gano el jugador %d!\n", jugador);
                terminado = 1;
            }
            else
            {
                if(jugador == 1)
                    jugador = 2;
                else
                    jugador = 1;
            }
        }
    }

    printf("Fin del juego.\n");
    return 0;
}
